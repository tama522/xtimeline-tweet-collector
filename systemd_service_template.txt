# Xタイムライン自動更新 systemdサービステンプレート
# 使用方法: /etc/systemd/system/xtimeline.service としてコピーし、パスを適切に変更してください

[Unit]
Description=X Timeline Auto Update Service
After=network.target
Wants=network-online.target

[Service]
Type=simple
User=your_username
Group=your_groupname
WorkingDirectory=/path/to/xtimeline
ExecStart=/usr/bin/python3 /path/to/xtimeline/auto_update.py
Restart=always
RestartSec=30

# 環境変数の設定
Environment=PYTHONPATH=/path/to/xtimeline
Environment=PYTHONUNBUFFERED=1

# セキュリティ設定
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ReadWritePaths=/path/to/xtimeline

# ログ設定
StandardOutput=journal
StandardError=journal
SyslogIdentifier=xtimeline

[Install]
WantedBy=multi-user.target
